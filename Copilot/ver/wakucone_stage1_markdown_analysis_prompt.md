```python
import pandas as pd

def read_csv_safely(file_path):
    encodings = ["utf-8-sig", "utf-8", "cp932", "shift_jis"]
    last_error = None

    for enc in encodings:
        try:
            return pd.read_csv(file_path, encoding=enc)
        except Exception as e:
            last_error = e

    raise last_error
```

```python
def extract_yyyymm(file_name):

    if match:
        return match.group(1)
    return None
```

## 10-5. 共通集計

対象人数、対象端末数、対象グループ数は、存在するCSVから重複を除いて算出してください。

```python
def count_unique_values(dfs):
    names = set()
    computers = set()
    groups = set()

    for df in dfs.values():
        if "ユーザー名" in df.columns:
            names.update(df["ユーザー名"].dropna().astype(str).unique())
        if "コンピューター名" in df.columns:
            computers.update(df["コンピューター名"].dropna().astype(str).unique())
        if "グループ名" in df.columns:
            groups.update(df["グループ名"].dropna().astype(str).unique())

    return {
        "対象ユーザー数": len(names),
        "対象端末数": len(computers),
        "対象グループ数": len(groups)
    }
```

# 12. セキュリティ分析の論点
